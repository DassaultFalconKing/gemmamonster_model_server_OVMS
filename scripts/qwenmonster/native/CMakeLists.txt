cmake_minimum_required(VERSION 3.20)
project(r1_guard_matrix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(XGRAMMAR_SOURCE_DIR "" CACHE PATH "Sealed source snapshot")
set(RUNTIME_HEADERS_DIR "" CACHE PATH "Sealed runtime headers/import libraries")
set(XGRAMMAR_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
set(XGRAMMAR_BUILD_CXX_TESTS OFF CACHE BOOL "" FORCE)
set(XGRAMMAR_ENABLE_CPPTRACE OFF CACHE BOOL "" FORCE)
set(XGRAMMAR_ENABLE_COVERAGE OFF CACHE BOOL "" FORCE)
add_subdirectory("${XGRAMMAR_SOURCE_DIR}" xgrammar EXCLUDE_FROM_ALL)
add_executable(r1_guard_matrix matrix.cpp)
target_include_directories(r1_guard_matrix PRIVATE "${XGRAMMAR_SOURCE_DIR}/cpp")
target_link_libraries(r1_guard_matrix PRIVATE xgrammar)
add_executable(r1_original_checks original-checks.cpp)
target_compile_definitions(r1_original_checks PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(r1_original_checks PRIVATE "${XGRAMMAR_SOURCE_DIR}/cpp" "${RUNTIME_HEADERS_DIR}/include")
target_link_libraries(r1_original_checks PRIVATE xgrammar
  "${RUNTIME_HEADERS_DIR}/lib/openvino_genai.lib"
  "${RUNTIME_HEADERS_DIR}/lib/openvino.lib")
if(MSVC)
  target_compile_options(r1_guard_matrix PRIVATE /bigobj)
  target_compile_options(r1_original_checks PRIVATE /bigobj)
endif()
